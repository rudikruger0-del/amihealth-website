function printAiReport() { if (!REPORT_DATA) { alert("Report data not loaded yet."); return; } const ai = safeParseAI(REPORT_DATA.ai_results); const canonical = getCanonicalFromAI(ai) || {}; const routes = ai.routes || ai.route || (ai.ai_results && ai.ai_results.routes) || {}; const summaryImpression = routes.summary || (ai.summary && ai.summary.impression) || "No overall impression provided."; const nextSteps = routes.next_steps || ai.summary && ai.summary.suggested_follow_up || []; const createdAt = REPORT_DATA.created_at ? new Date(REPORT_DATA.created_at).toLocaleString() : ""; const verifyUrl = `https://amihealth.co.za/verify?id=${encodeURIComponent(REPORT_DATA.id)}`; const qrUrl = `https://api.qrserver.com/v1/create-qr-code/?size=90x90&data=${encodeURIComponent(verifyUrl)}`; // helper to decide flag string and flag class (text-only colours) function flagForItem(it) { if (!it || it.value == null) return { flagText: "—", flagClass: "flag-unknown" }; const low = it.reference_low != null ? Number(it.reference_low) : null; const high = it.reference_high != null ? Number(it.reference_high) : null; const val = Number(it.value); if (low != null && high != null) { if (!isNaN(val) && val < low) return { flagText: "low", flagClass: "flag-low" }; if (!isNaN(val) && val > high) return { flagText: "high", flagClass: "flag-high" }; return { flagText: "normal", flagClass: "flag-normal" }; } // fallback to any provided flag field const rawFlag = it.flag || it.flags || it.status || ""; if (typeof rawFlag === "string") { const rf = rawFlag.toLowerCase(); if (rf.includes("high") || rf === "h") return { flagText: "high", flagClass: "flag-high" }; if (rf.includes("low") || rf === "l") return { flagText: "low", flagClass: "flag-low" }; if (rf === "normal" || rf === "n") return { flagText: "normal", flagClass: "flag-normal" }; if (rf) return { flagText: rawFlag, flagClass: "flag-unknown" }; } return { flagText: "—", flagClass: "flag-unknown" }; } // build cbc rows from canonical or legacy array let cbcRowsHtml = ""; if (Object.keys(canonical).length > 0) { let idx = 1; for (const k of Object.keys(canonical)) { const it = canonical[k]; const val = it.value != null ? it.value : ""; const units = it.units || ""; const refLow = it.reference_low != null ? it.reference_low : ""; const refHigh = it.reference_high != null ? it.reference_high : ""; const refRange = (refLow || refHigh) ? `${refLow}–${refHigh}` : ""; const flag = flagForItem(it); cbcRowsHtml += ` ${idx++} ${k} ${val} ${units} ${refRange} ${flag.flagText} `; } } else { const legacy = ai.cbc || []; cbcRowsHtml = legacy.map((item, idx) => { const refLow = item.reference_low != null ? item.reference_low : ""; const refHigh = item.reference_high != null ? item.reference_high : ""; const refRange = refLow || refHigh ? `${refLow}–${refHigh}` : ""; const it = { value: item.value, reference_low: item.reference_low, reference_high: item.reference_high, flag: item.flag }; const flag = flagForItem(it); return ` ${idx + 1} ${item.analyte || ""} ${item.value != null ? item.value : ""} ${item.units || ""} ${refRange} ${flag.flagText} `; }).join(""); } const name = REPORT_DATA.name || "Unknown"; const age = REPORT_DATA.age != null ? REPORT_DATA.age : "–"; const sex = REPORT_DATA.sex || "–"; const win = window.open("", "_blank"); win.document.write(` AMI Health — AI Interpretation Report

${verifyUrl}

`); win.document.close(); win.focus(); win.print(); }
