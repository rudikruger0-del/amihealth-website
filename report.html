async function loadReport() { const params = new URLSearchParams(window.location.search); const id = params.get("id"); const res = await fetch(`/api/get-ai-report?id=${id}`); const data = await res.json(); if (!data || !data.report) { document.querySelector("#aiText").textContent = "Unable to load report."; return; } const rpt = data.report; // PDF via signed URL const pdfRes = await fetch("/api/get-file-url", { method: "POST", body: JSON.stringify({ file_path: rpt.file_path }), }); const pdfJson = await pdfRes.json(); document.querySelector("#pdfFrame").src = pdfJson.url; // Patient Info document.querySelector("#patientInfo").innerHTML = ` Name: ${rpt.name || "-"}
Age: ${rpt.age || "-"}
Sex: ${rpt.sex || "-"}
Date: ${new Date(rpt.created_at).toLocaleString()} `; // AI text document.querySelector("#aiText").textContent = rpt.ai_results || "No AI output."; } loadReport();
